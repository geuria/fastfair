function getCarrinho() { return JSON.parse(localStorage.getItem("carrinho")) || []; } function salvarCarrinho(carrinho) { localStorage.setItem("carrinho", JSON.stringify(carrinho)); } function adicionarAoCarrinho(produto) { const carrinho = getCarrinho(); const item = carrinho.find(p => p.id === produto.id); if (item) { item.quantidade += produto.quantidade; } else { carrinho.push(produto); } salvarCarrinho(carrinho); alert("Produto adicionado!"); }
